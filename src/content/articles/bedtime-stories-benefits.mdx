  <div class="mistake-item">

  <div class="mistake-item">

      <h2>اختر القصة معاً</h2>

  </div>

  <div class="step-card">

  </div>

  <div class="step-card">
    <div class="step-num">3</div>
    <div class="step-content">

  </div>

  <div class="step-card">
    <div class="step-num">4</div>

  </div>

  <div class="step-card">
    <div class="step-num">5</div>
